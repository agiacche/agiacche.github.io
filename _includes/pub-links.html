{% comment %} Renders the row of action links (arXiv, DOI, PDF, open access, code) for a publication. Single canonical place for this logic — used by the publications list, the individual publication page, and the CV. Usage: {% include pub-links.html post=post %} {% endcomment %} {% assign post = include.post %} {% assign has_links = false %} {% if post.arxiv or post.doi or post.pdf or post.open or post.git or post.slides %}{% assign has_links = true %}{% endif %} {% if has_links %} {% if post.arxiv %} arXiv {% endif %} {% if post.doi %} DOI {% endif %} {% if post.pdf %} PDF {% endif %} {% if post.open %} Open access {% endif %} {% if post.git %} Code {% endif %} {% if post.slides %} Slides {% endif %} {% endif %}
